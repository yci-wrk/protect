// ===================================================== // 受付カメラ 受け口スクリプト（Google Apps Script） // フォルダIDを2つ設定してください // ===================================================== const FOLDER_ID = ""; // 保存用：写真と情報txtが入る場所 const DISPLAY_FOLDER_ID = ""; // 表示用：6が最新3枚を取りに行く場所（常時3枚だけ置く想定） // ===================================================== // GitHub Pages から「保存」で呼ばれる（送るだけ） function doPost(e) { try { const body = JSON.parse(e.postData.contents); const folder = DriveApp.getFolderById(FOLDER_ID); if (body.type === "info") { // 情報を txt（プレーンテキスト）で保存 folder.createFile( body.name || (Date.now() + "_info.txt"), body.text || "", "text/plain" ); } else if (body.type === "photo") { const bytes = Utilities.base64Decode(body.image); const blob = Utilities.newBlob(bytes, "image/jpeg", body.name || (Date.now() + "_photo.jpg")); folder.createFile(blob); } return ContentService.createTextOutput("ok"); } catch (err) { return ContentService.createTextOutput("error: " + err); } } // GitHub Pages から「最新3枚取得」で呼ばれる（JSONPで返す） function doGet(e) { const callback = e.parameter.callback || "callback"; try { const type = e.parameter.type; if (type === "latest") { const count = parseInt(e.parameter.count || "3", 10); const result = getDisplayImages(count); return jsonpOut(callback, { images: result }); } return jsonpOut(callback, { images: [] }); } catch (err) { return jsonpOut(callback, { error: String(err) }); } } // 表示用フォルダの画像を「名前順」に並べ、後ろから count 枚を返す function getDisplayImages(count) { const folder = DriveApp.getFolderById(DISPLAY_FOLDER_ID); const files = folder.getFiles(); const images = []; while (files.hasNext()) { const f = files.next(); const mime = f.getMimeType(); if (mime && mime.indexOf("image/") === 0) { images.push({ name: f.getName(), id: f.getId(), mime: mime }); } } images.sort((a, b) => a.name < b.name ? -1 : (a.name > b.name ? 1 : 0)); const last = images.slice(Math.max(0, images.length - count)); return last.map(item => { const blob = DriveApp.getFileById(item.id).getBlob(); const b64 = Utilities.base64Encode(blob.getBytes()); return { name: item.name, dataUrl: "data:" + item.mime + ";base64," + b64 }; }); } function jsonpOut(callback, obj) { const out = callback + "(" + JSON.stringify(obj) + ");"; return ContentService.createTextOutput(out) .setMimeType(ContentService.MimeType.JAVASCRIPT); }
